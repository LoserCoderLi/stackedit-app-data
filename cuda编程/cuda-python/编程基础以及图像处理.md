```
import cv2

import numpy as np

from numba import cuda

import time

import math

  

# GPU Function

@cuda.jit

def process_gpu(img,channels):

tx = cuda.blockIdx.x*cuda.blockDim.x+cuda.threadIdx.x

ty = cuda.blockIdx.y*cuda.blockDim.y+cuda.threadIdx.y

for c in range(channels):

color = img[tx,ty][c]*2.0 + 30

if color>255:

img[tx, ty][c] = 255

elif color < 0:

img[tx, ty][c] = 0

else:

img[tx, ty][c] = int(color)

  

# CPU Function

def process_cpu(img,dst):

rows,cols,channel = img.shape

for i in range(rows):

for j in range(cols):

for c in range(channel):

color = img[i,j][c]*2.0 + 30

if color>255:

dst[i,j][c] = 255

elif color < 0:

dst[i,j][c] = 0

else:

dst[i,j][c] = int(color)

if __name__ == "__main__":

# 读取图片

img = cv2.imread("./images/demo.jpg")

rows,cols,channel = img.shape

dst_cpu = img.copy()

dst_gpu = img.copy()

start_cpu = time.time()

process_cpu(img,dst_cpu)

end_cpu = time.time()

time_cpu = end_cpu - start_cpu

print("cpu time:", time_cpu)

# GPU function

dImg = cuda.to_device(img)

treadsperblock = (16,16)

bolckspergird_x = int(math.ceil(rows/treadsperblock[0]))

bolckspergird_y = int(math.ceil(cols/treadsperblock[1]))

bolckspergird = (bolckspergird_x, bolckspergird_y)

cuda.synchronize()

start_gpu = time.time()

process_gpu[bolckspergird, treadsperblock](dImg, channel)

cuda.synchronize()

end_gpu = time.time()

print("gpu time:", end_gpu-start_gpu)

# 返回给CPU

dst_gpu = dImg.copy_to_host()

cv2.imwrite("result_cpu.jpg", dst_cpu)

cv2.imwrite("result_Gpu.jpg", dst_gpu)

print("Done")
```
